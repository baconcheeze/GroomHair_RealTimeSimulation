cmake_minimum_required(VERSION 3.24)
project(HairLabStrandBlockCPU LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "Separate CPU diagnostic tree" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." native)
add_executable(StrandBlockExperiment StrandBlockExperiment.cpp ../../src/RestData.cpp)
target_compile_features(StrandBlockExperiment PRIVATE cxx_std_20)
target_compile_options(StrandBlockExperiment PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(StrandBlockExperiment PRIVATE GuideOrientation)
set_target_properties(StrandBlockExperiment PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
