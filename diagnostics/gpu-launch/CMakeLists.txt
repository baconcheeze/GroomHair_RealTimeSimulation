cmake_minimum_required(VERSION 3.24)
project(HairG64Launch LANGUAGES CXX)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native-ls" cpu)
file(GLOB _paths "C:/Program Files (x86)/Windows Kits/10/bin/*/x64")
list(SORT _paths COMPARE NATURAL ORDER DESCENDING)
find_program(LAUNCH_DXC NAMES dxc.exe HINTS ${_paths} REQUIRED)
execute_process(COMMAND "${LAUNCH_DXC}" --version OUTPUT_VARIABLE _version COMMAND_ERROR_IS_FATAL ANY)
set(_compiled "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/shaders")
file(GENERATE OUTPUT "${_compiled}/compiler.txt" CONTENT "DXC=${LAUNCH_DXC}\n${_version}\nProfile=cs_6_0\nFlags=-HV 2021 -Gis $<IF:$<CONFIG:Debug>,-Od -Zi -Qembed_debug,-O3>\n")
foreach(_group 1 32 64 legacy)
 foreach(_entry Form Factor Initial Correct)
  foreach(_check 0 1)
   if(_check AND (_entry STREQUAL Form OR _group STREQUAL legacy))
    continue()
   endif()
   set(_source Packed.hlsl)
   set(_suffix g${_group})
   set(_define -DGROUP_SIZE=${_group})
   if(_group STREQUAL legacy)
    set(_source Legacy.hlsl)
    set(_suffix legacy)
    set(_define)
   endif()
   if(_check)
    list(APPEND _define -DOWNERSHIP_CHECK=1)
    set(_suffix check_${_suffix})
   endif()
   add_custom_command(OUTPUT "${_compiled}/${_entry}_${_suffix}.dxil"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${_compiled}"
    COMMAND "${LAUNCH_DXC}" -T cs_6_0 -HV 2021 -Gis "$<IF:$<CONFIG:Debug>,-Od,-O3>" "$<$<CONFIG:Debug>:-Zi>" "$<$<CONFIG:Debug>:-Qembed_debug>" ${_define} -E ${_entry} -Fo "${_compiled}/${_entry}_${_suffix}.dxil" -Fc "${_compiled}/${_entry}_${_suffix}.ll" "${CMAKE_CURRENT_LIST_DIR}/shaders/${_source}"
    DEPENDS "shaders/${_source}" shaders/DoubleMath.hlsli COMMAND_EXPAND_LISTS VERBATIM)
   list(APPEND _outputs "${_compiled}/${_entry}_${_suffix}.dxil")
  endforeach()
 endforeach()
endforeach()
add_custom_target(LaunchShaders DEPENDS ${_outputs})
add_executable(G64LaunchCandidate Main.cpp Engine.cpp Ownership.cpp Study.cpp Hybrid.generated.cpp HybridBridge.generated.cpp HybridSuite.cpp ../../src/AutonomousReference.cpp ../../src/RestData.cpp)
target_compile_features(G64LaunchCandidate PRIVATE cxx_std_20)
target_compile_options(G64LaunchCandidate PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(G64LaunchCandidate PRIVATE NativeStructuredLS dxguid)
set_target_properties(G64LaunchCandidate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_dependencies(G64LaunchCandidate LaunchShaders)
add_executable(G64LaunchMemory Memory.cpp Engine.cpp)
target_compile_features(G64LaunchMemory PRIVATE cxx_std_20)
target_compile_options(G64LaunchMemory PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(G64LaunchMemory PRIVATE NativeStructuredLS dxguid)
set_target_properties(G64LaunchMemory PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_dependencies(G64LaunchMemory LaunchShaders)
