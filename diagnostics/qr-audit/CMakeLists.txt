cmake_minimum_required(VERSION 3.24)
project(HairLabQRAuditCPU LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "Separate CPU diagnostic tree" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." native)
add_executable(QRFreeze QRFreeze.cpp ../../src/RestData.cpp)
target_compile_features(QRFreeze PRIVATE cxx_std_20)
target_compile_options(QRFreeze PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(QRFreeze PRIVATE GuideOrientation)
set_target_properties(QRFreeze PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(QRMetricProbe MetricProbe.cpp)
target_compile_features(QRMetricProbe PRIVATE cxx_std_20)
target_compile_options(QRMetricProbe PRIVATE /W4 /fp:precise)
set_target_properties(QRMetricProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
