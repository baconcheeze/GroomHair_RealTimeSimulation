cmake_minimum_required(VERSION 3.24)
project(HairLabLSRefinementCPU LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "Separate CPU diagnostic tree" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." native)
add_executable(LSFactorHost FactorHost.cpp ../../src/RestData.cpp)
target_compile_features(LSFactorHost PRIVATE cxx_std_20)
target_compile_options(LSFactorHost PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(LSFactorHost PRIVATE GuideOrientation)
set_target_properties(LSFactorHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(LSExportSuite ExportSuite.cpp ../../src/RestData.cpp)
target_compile_features(LSExportSuite PRIVATE cxx_std_20)
target_compile_options(LSExportSuite PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(LSExportSuite PRIVATE GuideOrientation)
set_target_properties(LSExportSuite PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(LSRefinedReference Nonlinear.cpp ../../src/RestData.cpp)
target_compile_features(LSRefinedReference PRIVATE cxx_std_20)
target_compile_options(LSRefinedReference PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(LSRefinedReference PRIVATE GuideOrientation)
set_target_properties(LSRefinedReference PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
