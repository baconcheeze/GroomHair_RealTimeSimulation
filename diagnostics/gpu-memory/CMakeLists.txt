cmake_minimum_required(VERSION 3.24)
project(HairG64Memory LANGUAGES CXX)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native-ls" cpu)
file(GLOB _paths "C:/Program Files (x86)/Windows Kits/10/bin/*/x64")
list(SORT _paths COMPARE NATURAL ORDER DESCENDING)
find_program(MEMORY_DXC NAMES dxc.exe HINTS ${_paths} REQUIRED)
execute_process(COMMAND "${MEMORY_DXC}" --version OUTPUT_VARIABLE _version COMMAND_ERROR_IS_FATAL ANY)
set(_compiled "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/shaders")
file(GENERATE OUTPUT "${_compiled}/compiler.txt" CONTENT "DXC=${MEMORY_DXC}\n${_version}\nProfile=cs_6_0\nFlags=-HV 2021 -Gis $<IF:$<CONFIG:Debug>,-Od -Zi -Qembed_debug,-O3>\n")
foreach(_layout base work both)
 foreach(_entry Form Factor Initial Correct)
  foreach(_check 0 1)
   if(_check AND _entry STREQUAL Form)
    continue()
   endif()
   set(_defines -DGROUP_SIZE=32)
   if(NOT _layout STREQUAL base)
    list(APPEND _defines -DWORK32=1)
   endif()
   if(_layout STREQUAL both)
    list(APPEND _defines -DINPUT32=1)
   endif()
   set(_suffix ${_layout})
   if(_check)
    list(APPEND _defines -DOWNERSHIP_CHECK=1)
    set(_suffix check_${_layout})
   endif()
   add_custom_command(OUTPUT "${_compiled}/${_entry}_${_suffix}.dxil"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${_compiled}"
    COMMAND "${MEMORY_DXC}" -T cs_6_0 -HV 2021 -Gis "$<IF:$<CONFIG:Debug>,-Od,-O3>" "$<$<CONFIG:Debug>:-Zi>" "$<$<CONFIG:Debug>:-Qembed_debug>" ${_defines} -E ${_entry} -Fo "${_compiled}/${_entry}_${_suffix}.dxil" -Fc "${_compiled}/${_entry}_${_suffix}.ll" "${CMAKE_CURRENT_LIST_DIR}/shaders/Packed.hlsl"
    DEPENDS shaders/Packed.hlsl shaders/DoubleMath.hlsli COMMAND_EXPAND_LISTS VERBATIM)
   list(APPEND _outputs "${_compiled}/${_entry}_${_suffix}.dxil")
  endforeach()
 endforeach()
endforeach()
add_custom_target(MemoryShaders DEPENDS ${_outputs})
add_executable(G64MemoryCandidate Main.cpp Engine.cpp Codec.cpp Ownership.cpp Study.cpp Hybrid.generated.cpp HybridBridge.generated.cpp HybridSuite.cpp ../../src/AutonomousReference.cpp ../../src/RestData.cpp)
target_compile_features(G64MemoryCandidate PRIVATE cxx_std_20)
target_compile_options(G64MemoryCandidate PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(G64MemoryCandidate PRIVATE NativeStructuredLS dxguid)
set_target_properties(G64MemoryCandidate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_dependencies(G64MemoryCandidate MemoryShaders)
add_executable(G64CodecAudit CodecAudit.cpp)
target_compile_features(G64CodecAudit PRIVATE cxx_std_20)
target_compile_options(G64CodecAudit PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(G64CodecAudit PRIVATE NativeStructuredLS dxguid)
set_target_properties(G64CodecAudit PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(G64BatchOutputAudit BatchOutputAudit.cpp Engine.cpp)
target_compile_features(G64BatchOutputAudit PRIVATE cxx_std_20)
target_compile_options(G64BatchOutputAudit PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(G64BatchOutputAudit PRIVATE NativeStructuredLS dxguid)
set_target_properties(G64BatchOutputAudit PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
