cmake_minimum_required(VERSION 3.24)
project(HairLabNativeLS LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "Isolated CPU candidate" FORCE)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This diagnostic QD configuration requires Windows x64 MSVC")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." native)
add_library(NativeStructuredLS STATIC NativeLS.cpp)
target_compile_features(NativeStructuredLS PUBLIC cxx_std_20)
target_include_directories(NativeStructuredLS PUBLIC ../../third_party/qd-2.3.24/include)
target_compile_options(NativeStructuredLS PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(NativeStructuredLS PUBLIC GuideOrientation)
add_executable(NativeLSCandidate Main.cpp Bridge.cpp Benchmark.cpp ../../src/RestData.cpp)
target_compile_features(NativeLSCandidate PRIVATE cxx_std_20)
target_compile_options(NativeLSCandidate PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(NativeLSCandidate PRIVATE NativeStructuredLS)
set_target_properties(NativeLSCandidate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
