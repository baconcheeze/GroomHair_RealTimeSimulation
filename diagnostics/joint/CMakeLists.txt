cmake_minimum_required(VERSION 3.24)
project(HairLabJointCPU LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "CPU diagnostic tree only" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." native)
add_executable(JointPrototype JointPrototype.cpp ../../src/RestData.cpp)
target_compile_features(JointPrototype PRIVATE cxx_std_20)
target_compile_options(JointPrototype PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(JointPrototype PRIVATE GuideOrientation)
set_target_properties(JointPrototype PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
