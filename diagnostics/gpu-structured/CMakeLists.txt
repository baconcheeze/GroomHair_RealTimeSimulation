cmake_minimum_required(VERSION 3.24)
project(HairGpuStructuredPilot LANGUAGES CXX)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native-ls" cpu)
add_executable(GpuStructuredProbe Probe.cpp ../../src/RestData.cpp)
target_compile_features(GpuStructuredProbe PRIVATE cxx_std_20)
target_compile_options(GpuStructuredProbe PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(GpuStructuredProbe PRIVATE NativeStructuredLS dxguid)
set_target_properties(GpuStructuredProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(GLOB _paths "C:/Program Files (x86)/Windows Kits/10/bin/*/x64")
list(SORT _paths COMPARE NATURAL ORDER DESCENDING)
find_program(PILOT_DXC NAMES dxc.exe HINTS ${_paths} REQUIRED)
execute_process(COMMAND "${PILOT_DXC}" --version OUTPUT_VARIABLE _version)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/shaders/compiler.txt" CONTENT "DXC=${PILOT_DXC}\n${_version}\nProfile=cs_6_0\nFlags=-HV 2021 -Gis $<IF:$<CONFIG:Debug>,-Od -Zi -Qembed_debug,-O3>\n")
set(_compiled "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/shaders")
foreach(_name Probe UnsafeProbe)
 set(_define)
 if(_name STREQUAL UnsafeProbe)
  set(_define -DUNSAFE_SQRT=1)
 endif()
 add_custom_command(OUTPUT "${_compiled}/${_name}.dxil"
 COMMAND ${CMAKE_COMMAND} -E make_directory "${_compiled}"
 COMMAND "${PILOT_DXC}" -T cs_6_0 -HV 2021 -Gis "$<IF:$<CONFIG:Debug>,-Od,-O3>" "$<$<CONFIG:Debug>:-Zi>" "$<$<CONFIG:Debug>:-Qembed_debug>" ${_define} -E Probe -Fo "${_compiled}/${_name}.dxil" -Fc "${_compiled}/${_name}.ll" "${CMAKE_CURRENT_LIST_DIR}/shaders/Probe.hlsl"
 DEPENDS shaders/Probe.hlsl shaders/DoubleMath.hlsli COMMAND_EXPAND_LISTS VERBATIM)
 list(APPEND _probe_outputs "${_compiled}/${_name}.dxil")
endforeach()
add_custom_target(PilotProbeShaders DEPENDS ${_probe_outputs})
add_dependencies(GpuStructuredProbe PilotProbeShaders)
foreach(_entry Form Factor Initial Correct)
 foreach(_mixed 0 1)
  if(_entry STREQUAL Form AND _mixed)
   continue()
  endif()
  set(_name ${_entry})
  set(_define)
  if(_mixed)
   set(_name ${_entry}32)
   set(_define -DMIXED=1)
  endif()
  add_custom_command(OUTPUT "${_compiled}/${_name}.dxil"
   COMMAND ${CMAKE_COMMAND} -E make_directory "${_compiled}"
   COMMAND "${PILOT_DXC}" -T cs_6_0 -HV 2021 -Gis "$<IF:$<CONFIG:Debug>,-Od,-O3>" "$<$<CONFIG:Debug>:-Zi>" "$<$<CONFIG:Debug>:-Qembed_debug>" ${_define} -E ${_entry} -Fo "${_compiled}/${_name}.dxil" -Fc "${_compiled}/${_name}.ll" "${CMAKE_CURRENT_LIST_DIR}/shaders/Structured.hlsl"
   DEPENDS shaders/Structured.hlsl shaders/DoubleMath.hlsli COMMAND_EXPAND_LISTS VERBATIM)
  list(APPEND _solver_outputs "${_compiled}/${_name}.dxil")
 endforeach()
endforeach()
add_custom_target(PilotSolverShaders DEPENDS ${_solver_outputs})
add_executable(GpuStructuredLSCandidate Main.cpp Engine.cpp Hybrid.generated.cpp HybridBridge.generated.cpp HybridSuite.cpp Batch.cpp ../../src/AutonomousReference.cpp ../../src/RestData.cpp)
target_compile_features(GpuStructuredLSCandidate PRIVATE cxx_std_20)
target_compile_options(GpuStructuredLSCandidate PRIVATE /W4 /permissive- /utf-8 /fp:precise)
target_link_libraries(GpuStructuredLSCandidate PRIVATE NativeStructuredLS dxguid)
set_target_properties(GpuStructuredLSCandidate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_dependencies(GpuStructuredLSCandidate PilotSolverShaders)

target_link_libraries(GpuStructuredMemory PRIVATE NativeStructuredLS dxguid)
set_target_properties(GpuStructuredMemory PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
